23class MouGenerator(abc.ABC): 

24 """Abstract base for MOU PDF generators. 

25 

26 Subclasses declare ``_template_name`` and implement ``_build_context``. 

27 ``generate()`` is the template method: it assembles the common context, 

28 delegates type-specific context to the subclass, then renders the PDF. 

29 """ 

30 

31 _template_name: str 

32 

33 def __init__(self, director_name, director_title, director_signature_b64): 

34 self._director_name = director_name 

35 self._director_title = director_title 

36 self._director_signature_b64 = director_signature_b64 

37 

38 def generate(self, first_name, last_name, project_name, **kwargs) -> bytes: 

39 context = self._common_context(first_name, last_name, project_name) 

40 self._build_context(context, **kwargs) 

41 return self._render(context) 

42 

43 @abc.abstractmethod 

44 def _build_context(self, context: dict, **kwargs) -> None: 

45 """Populate ``context`` with type-specific template variables.""" 

46 

47 def _common_context(self, first_name, last_name, project_name) -> dict: 

48 return { 

49 "brc_logo_b64": _brc_logo_b64, 

50 "director_name": self._director_name, 

51 "director_title": self._director_title, 

52 "director_signature_b64": f"data:image/png;base64,{self._director_signature_b64}", 

53 "pi_name": f"{first_name} {last_name}", 

54 "date": datetime.date.today().strftime("%B %d, %Y"), 

55 "project": project_name, 

56 } 

57 

58 def _render(self, context: dict) -> bytes: 

59 html = template_env.get_template(self._template_name).render(context) 

60 

61 # Playwright's subprocess transport calls sys.stderr.fileno() when 

62 # spawning the Node.js driver process. Under mod_wsgi, sys.stderr is 

63 # replaced with a log object that has no real file descriptor, raising 

64 # OSError. Temporarily substitute a real file so the fd lookup succeeds. 

65 _orig_stderr = sys.stderr 

66 sys.stderr = open(os.devnull, "w") 

67 try: 

68 with sync_playwright() as p: 

69 with p.chromium.launch( 

70 args=[ 

71 "--no-sandbox", # required in Docker / restricted-namespace VMs 

72 "--disable-dev-shm-usage", # avoids /dev/shm OOM in containers 

73 ] 

74 ) as browser: 

75 page = browser.new_page() 

76 page.set_content(html, wait_until="networkidle") 

77 pdf = page.pdf( 

78 format="Letter", 

79 print_background=True, 

80 scale=0.82, 

81 margin={ 

82 "top": "0.4in", 

83 "right": "0.4in", 

84 "bottom": "0.4in", 

85 "left": "0.4in", 

86 }, 

87 ) 

88 finally: 

89 sys.stderr.close() 

90 sys.stderr = _orig_stderr 

91 

92 return pdf
